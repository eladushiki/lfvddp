\subsection{Testing Hypotheses}

We now look at two different dataset with some underlying physical parameters that differentiate them. We want to formulate the question - whether or not these two sets stem from the same underlying probability distribution? For starters, we denote them as $\meas{A}$ and $\meas{B}$. Each of our hypotheses would consist of a pair of (parametrized) probability distributions, allowing us to further compare them by likelihood measures.

\subsubsection{Hypotheses in the Weighted Setting}
We now define the null and alternative hypotheses using our reconstructed dataset formalism:

\[
\begin{aligned}
\nullhypothesis &: \quad
\begin{cases}
    {\dist{\reco{A}}^0} = \dfrac{N_\reco{A}}{Z_h} e^{h(\bar{x})} \refdist \\
    {\dist{\reco{B}}^0} = \dfrac{N_\reco{B}}{Z_h} e^{h(\bar{x})} \refdist
\end{cases} \\
\hypothesis{1} &: \quad
\begin{cases}
    {\dist{\reco{A}}^1} = \dfrac{N_\reco{A}}{Z_f} e^{f(\bar{x})} \refdist \\
    {\dist{\reco{B}}^1} = \dfrac{N_\reco{B}}{Z_g} e^{(f-g)(\bar{x})} \refdist
\end{cases}
\end{aligned}
\]

Here \(n_{\dataset{A}}(x)\) and \(n_{\dataset{B}}(x)\) are estimated probability distributions that may or may not differ from each other. We now say that $h$ ($f$) in the null (signal) hypothesis is constructed from parameters that are as arbitrary as $\refdist[\anyevent]$, and $g$ contains the parameters of interest. It can be restated that

\begin{equation}
\begin{aligned}
    \nullhypothesis(h)(\anyevent) &\to \nullhypothesis(\nparam)(\anyevent) \\
    \hypothesis{1}(f, g)(\anyevent) &\to \hypothesis{1}(\param, \nparam)(\anyevent) \\
    (\hypothesis{1}|\param = 0) &= \nullhypothesis
\end{aligned}
\end{equation}

Our estimation for the appearing probability densities of the datasets would be the reweighed datasets themselves. Then, we can get a likelihood value for each estimation stated in our hypotheses from \ref{eq:reconstructed_log_likelihood_of_a_distribution}. To estimate the likelihood of a hypothesis, we should multiply both its distributions' likelihoods

\begin{equation}
    \likelihood(\hypothesis{i} | \dataset{A}, \dataset{B}) = \likelihood(\dist{\dataset{A}}|\dataset{A})^i \cdot \likelihood(\dist{\dataset{B}}|\dataset{B})^i
\end{equation}

Then, to compare between the likelihood of both hypotheses, we construct (as in \cite{LastDDPPaper}) the test statistic to be:

\begin{equation}
\begin{aligned}
t &= 2 \log \left(
\frac{
\max_{\param, \nparam} (\likelihood(\hypothesis{1}(\param, \nparam)|\reco{A}, \reco{B}))
}{
\max_{\param = 0, \nparam} (\likelihood(\nullhypothesis(\param, \nparam)|\reco{A}, \reco{B}))
}\right)
\\
&= 2 \log \left(
\frac{
\max_{g, f} (\likelihood(\dist{\reco{A}}^1|\reco{A})\likelihood(\dist{\reco{B}}^1|\reco{B})\likelihood(\nparam)
}{
\max_{g = 0, h} (\likelihood(\dist{\reco{A}}^0|\reco{A})\likelihood(\dist{\reco{B}}^0|\reco{B}))
}\right)
\\
&= 2 \log \left(
\frac{ max_{f, g}(
e^{- \int \dist{\reco{A}}^1} \prod\limits_{x\in \reco{A}} \dist{\reco{A}}^1
\cdot
e^{- \int \dist{\reco{B}}^1} \prod\limits_{x\in \reco{B}} \dist{\reco{B}}^1
\cdot
\prod\limits_{\nparam} \frac{1}{2 \sigma_0^2} (\nparam)^2
)}{ max_{h}(
e^{- \int \dist{\reco{A}}^0} \prod\limits_{x\in \reco{A}} \dist{\reco{A}}^0
\cdot
e^{- \int \dist{\reco{B}}^0} \prod\limits_{x\in \reco{B}} \dist{\reco{B}}^0
)} \right)
\\
&= 2 \Big(
- N_{\reco{A}^1} + N_{\reco{A}^0} - N_{\reco{B}^1} + N_{\reco{B}^0} \\
&+ \sum\limits_{x\in\reco{A}} \left( \log (\maxdist{\reco{A}}^1) - \log (\maxdist{\reco{A}}^0) \right)
+
\sum\limits_{x\in \reco{B}} \left( \log (\maxdist{\reco{B}}^1) - \log (\maxdist{\reco{B}}^0) \right)
+
2\sum\limits_{\nparam} \log (\nparam)
\Big) + C
\\
&= 2 \Big(
- N_{\reco{A}^1} + N_{\reco{A}^0} - N_{\reco{B}^1} + N_{\reco{B}^0}
\\ &+
\sum\limits_{x\in\reco{A}} \left( \log (\dfrac{N_\reco{A}}{Z_f} e^{\hat{f}(\bar{x})} \refdist) - \log (\dfrac{N_\reco{A}}{Z_h} e^{\hat{h}(\bar{x})} \refdist) \right)
\\ &+
\sum\limits_{x\in \reco{B}} \left( \log (\dfrac{N_\reco{B}}{Z_g} e^{(\hat{f}-\hat{g})(\bar{x})} \refdist) - \log (\dfrac{N_\reco{B}}{Z_h} e^\hat{h}(\bar{x}) \refdist) \right)
+
2\sum\limits_{\nparam} \log (\nparam)
\Big) + C
\end{aligned}
\end{equation}

$$
\begin{aligned} \label{eq:def_test_statistic_with_nuisance}
t_{\dataset{A}+\dataset{B}}(\dataset{A}) &= -2 \log \big( \likelihood(\dataset{A}| \nu^j_m) \cdot \likelihood(\nu^j_m) \big) = -2 \log \likelihood(\dataset{A}|\nu^j_m) + \sum_{i=0,j=0}^{N_\reco{A}, k} \log \likelihood(\nu^j_m) \\
 &= -2 \cdot \min_{f(x)}
\left[
    - \frac{N_{\reco{A}}}{N_{\reco{A}} + N_{\reco{B}}}
    \sum_{x_i \in \reco{A} \cup \reco{B}} \frac{1}{\theta(x_i)} (e^{f(x_i)} - 1)
    + \sum_{x_i \in \reco{A}} \frac{1}{\theta(x_i)} f(x_i)
    - \sum_{\nu^j} \frac{1}{2 \sigma_0^2} (\nu^j_m)^2
\right],
\end{aligned}
$$

\noindent with a similar computation for $t_{\dataset{A}+\dataset{B}}(\dataset{B})$.

Therefore, the loss function can be expressed as

$$
\begin{aligned} \label{eq:loss_function_with_nuisance}
L[f, \nu^j] &=
    - \frac{N_{\reco{A}}}{N_{\reco{A}} + N_{\reco{B}}} \frac{1}{\theta(x_i)}
    \left[
        \sum_{x_i \in \reco{A} \cup \reco{B}}
        (1-y_i)(e^{f(x_i)} - 1)
        - y_i f(x_i)
    \right]
    - \sum_{\nu^j} \frac{1}{2 \sigma_0^2} (\nu^j)^2.
\end{aligned}
$$
