\section{Motivation}

\begin{frame}{Classic SM Analysis}
    \begin{itemize}
        \item General steps of a classical search for BSM physics:
        \begin{itemize}
            \item Hypothesize a change from the SM postulates
            \item Formulate production mechanisms and decay modes
            \item Construct a statistical model of the model behavior
            \item Monte Carlo simulation of the new physics in adetector
            \item Plan an experiment to magnify signal / background
            \item Estimate uncertainties
            \item Unblinding and analysis
        \end{itemize}
        \item The procedure is thorough, but labor intensive
        \item Eventually, this examined one specific cut among almost infinite possibilities
    \end{itemize}
\end{frame}

\begin{frame}{DDP}
    \begin{itemize}
        \item We propose and alternative - start from looking at the data
        \item Scan for deviation from what the SM predicts
        \item Direct model builders to explain found signals
    \end{itemize}
\end{frame}

\section{Overview}

\begin{frame}{LFV - DDP}
    \begin{itemize}
        \item The SM predicts an accidental global symmetry of the lepton flavor universality
        \item There is motivation to look for places in which this symmetry does not hold - driven by observed neutrino oscillations
        \item We'll use this symmetry assumption to compare pairs of equivalent datasets and look for differences without modelling
    \end{itemize}

    \end{itemize}

        \begin{equation}
        \begin{aligned}
            \nullhypothesis &: \quad
            \begin{cases}
                \pdf_{\dataset{A}}(\eventsymbol)
                &=
                \frac{N_{\meas{A}}}
                    {\int \refdist[\eventsymbol]\,dx}
                e^{h(\eventsymbol)} \refdist[\eventsymbol],
                \\
                \dist{\dataset{B}}
                &=
                \frac{N_{\meas{B}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{h(\eventsymbol)+r} \refdist[\eventsymbol],
            \end{cases} \\
            \hypothesis{1} &: \quad
            \begin{cases}
                \dist{\dataset{A}}
                &=
                \frac{N_{\meas{A}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{f(\eventsymbol)} \refdist[\eventsymbol],
                \\
                \dist{\dataset{B}}
                &=
                \frac{N_{\meas{B}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{g(\eventsymbol)} \refdist[\eventsymbol].
            \end{cases}
        \end{aligned}
        \end{equation}
    \end{block}
    \begin{itemize}
        \item Test statistic
    \end{itemize}
    \begin{equation}
        t = 2 \log \left(
            \dfrac{
                \max\limits_{f, g} \left(
                    \likelihood(\hypothesis{1}|\dataset{A}, \dataset{B})
                \right)
            }{
                \max\limits_{h, r} \left(
                    \likelihood(\nullhypothesis|\dataset{A}, \dataset{B})
                \right)
            }
        \right)
    \end{equation}

\section{Formulation}

\begin{frame}{Observing Events}
    \begin{itemize}
        \item We consider stochastic physical processes, to measure which we repeat the same experiment many times and gather statistics of the results.
        \item Each repetition is an event, and we gather such to create datasets, and represent them as some $k$ physical quantities that we measure them by:
    \end{itemize}
    \begin{equation}
        \event \in \mathbb{R}^k, \quad \event \sim \lambda, \quad \dataset{A} = \{\event\}
    \end{equation}
    \begin{itemize}
        \item with $\lambda$ some probability distribution, that we express as a shift of some common reference distribution
    \end{itemize}
    \begin{equation}
        \label{eq:parametrized_distribution}
        \parampdf(\anyevent) = e^{f_{\param}(\anyevent)}h_{\nparam}(\anyevent)\refdist
    \end{equation}
    \begin{itemize}
        \item where $\param$s are the parameters of interest that make $f$, and $\nparam$s are nuisance parameters that compose $g$.
    \end{itemize}
\end{frame}

\begin{frame}{Distribution Likelihood}
    \begin{itemize}
        \item Likelihood of a distribution given an observation $\dataset{A}$ is the probability of observing $\dataset{A}$ given the distribution
    \end{itemize}
    \begin{equation}
        \likelihood(\parampdf | \dataset{A}) = P(\dataset{A} | \parampdf)
    \end{equation}
    \begin{itemize}
        \item Likelihood of a dataset generated from Poisson statistics then becomes
    \end{itemize}
    \begin{equation}
        \likelihood(\parampdf | \dataset{A})
        = \dfrac{e^{-N_{\dataset{A}}(\param, \nparam)}}{N_{\dataset{A}}!}
        \prod\limits_{\event \in \dataset{A}} \parampdf(\event)
        = \dfrac{e^{-\int \parampdf(\anyevent) d\anyevent}}{N_{\dataset{A}}!}
        \prod\limits_{\event \in \dataset{A}} \parampdf(\event)
    \end{equation}
    \begin{itemize}
        \item which makes the log-likelihood of a distribution like in Eq.~\ref{eq:parametrized_distribution}
    \end{itemize}
    \begin{equation}
        \llikelihood( \parampdf | \dataset{A}) = -N_{\dataset{A}} + \sum_{\event \in \dataset{A}} \log\big({N}_{\dataset{A}} e^{\paramf(\event)} h_{\nparam}(\event)\refdist[\event]\big) + C_\dataset{A}
    \end{equation}
\end{frame}

\begin{frame}{Neymann - Pearson}
    \begin{lemma}
        \textbf{Neymann - Pearson:} The most discriminative test statistic for testing a signal hypothesis vs the null hypothesis, at a given significance level $\alpha$, is the test if the profile likelihood ratio is greater thatn some threshold $\eta$, defined as
    \end{lemma}
    \begin{equation}
        \Lambda(\mu) = \dfrac{
            \likelihood(\hat{\mu},\hat{\theta})
        }{
            \likelihood(\mu = 0, \hat{\hat{\theta}})
        } > \eta
    \end{equation}
    \begin{block}{}
        when $\eta$ is chosen such that the probability of rejecting the null hypothesis when it is true is $\alpha$:
    \end{block}
    \begin{equation}
        P(\Lambda(\mu) \geq \eta | \nullhypothesis \text{ is true}) = \alpha
    \end{equation}
\end{frame}

\begin{frame}{Observing Data}
    \begin{itemize}
        \item Events have to be registered in a detector in order to be observed.
        \item Detector does not always succeed in event construction, so we say it has efficiency, $\detector(\anyevent)$, that is bounded in $\left[0, 1\right]$ and may depend on the attributes of $\event$.
        \item We are about to compare similar datasets, hence interested only in the efficiency difference between two such, so we define
    \end{itemize}
    \begin{equation}
        \detector_{A}(\anyevent) - \detector_{B}(\anyevent) \coloneqq 2 \nparam(\anyevent)
    \end{equation}
    \begin{itemize}
        \item In other words, distributions can be parametrized again by an additional factor
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        \dist{\meas{A}} &\propto (1 + \nparam(\anyevent)) \\
        \dist{\meas{B}} &\propto (1 - \nparam(\anyevent))
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Data Regions}
    \begin{itemize}
        \item We consider two pairs of datasets, as follows;
    \end{itemize}

    \end{itemize}
\end{frame}

\begin{frame}{Hypotheses}
    \begin{itemize}
        \item We postulate two hypotheses, signal and null, as follows
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        \nullhypothesis |_{SR} &: \quad
        \begin{cases}
            {\srdist{\meas{A}}^0} = N_\srmeas{A} (1 + {\nparam(\anyevent)}) \srrefdist \\
            {\srdist{\meas{B}}^0} = N_\srmeas{B} (1 - {\nparam(\anyevent)}) \srrefdist
        \end{cases} \\
        \hypothesis{1} |_{SR} &: \quad
        \begin{cases}
            {\srdist{\meas{A}}^1} = N_\srmeas{A} e^{f(\anyevent)} (1 + {\nparam(\anyevent)}) \srrefdist \\
            {\srdist{\meas{B}}^1} = N_\srmeas{B} e^{g(\anyevent)} (1 - {\nparam(\anyevent)}) \srrefdist
        \end{cases} \\
        \nullhypothesis |_{CR} = \hypothesis{1} |_{CR} &: \quad
        \begin{cases}
            \crdist{\meas{A}}^i = N_\crmeas{A} (1 + {\nparam(\anyevent)}) \crrefdist \\
            \crdist{\meas{B}}^i = N_\crmeas{B} (1 - {\nparam(\anyevent)}) \crrefdist
        \end{cases}
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Likelihood and Test Statistics}
    \begin{itemize}
        \item Log likelihood of a hypothesis is a product of that of its parts
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        \likelihood(\hypothesis{i} | \meas{A}, \meas{B})
        &= \likelihood(\hypothesis{i}|\srmeas{A}, \srmeas{B}) \cdot \likelihood(\hypothesis{i}|\crmeas{A}, \crmeas{B}) \\&
        = \likelihood(\srdist{\meas{A}}^i|\srmeas{A}) \cdot \likelihood(\srdist{\meas{B}}^i|\srmeas{B}) \\&
        \cdot \likelihood(\crdist{\meas{A}}^i|\crmeas{A}) \cdot \likelihood(\crdist{\meas{B}}^i|\crmeas{B})
    \end{aligned}
    \end{equation}
    \begin{itemize}
        \item From which we build the test statistic from its definition ($\param \to f, g$, note that $\nullhypothesis = \hypothesis{1}(f, g = 0)$)
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        t &= 2 \log \left(
            \frac{
                \max\limits_{f, g, \nparam} (\likelihood(\hypothesis{1}|\meas{A}, \meas{B}, \nparam))
            }{
                \max\limits_{\nparam} (\likelihood(\nullhypothesis|\meas{A}, \meas{B}, \nparam))
        }\right) \\&
        = - 2 \left(
            {\min\limits_{f, g, \nparam}(-\log \likelihood(\hypothesis{1}|\meas{A}, \meas{B}, \nparam))} 
            - {\min\limits_{\nparam}(-\log \likelihood(\nullhypothesis|\meas{A}, \meas{B}, \nparam))}
        \right)
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Minimization by Loss Function}
    \begin{itemize}
        \item We reduce the problem of calculating this to the calculation of two minima
    \end{itemize}
    \begin{equation}
    \label{eq:formal_loss_functions}
    \begin{aligned}
        L^{n}[f, g, \nparam] &= \Bigg(
            \int (e^{f(\anyevent)}\srdist{\meas{A}}^0 + e^{g(\anyevent)}\srdist{\meas{B}}^0)
            + \int (\crdist{\meas{A}}^0 + \crdist{\meas{B}}^0)
            \\&
            - \sum\limits_{\event \in \srmeas{A}} f(\anyevent)
            - \sum\limits_{\event \in \srmeas{B}} g(\anyevent) \\&
            - \sum\limits_{\event \in \meas{A}} \log(1 + {\nparam(\anyevent)})
            - \sum\limits_{\event \in \meas{B}} \log(1 - {\nparam(\anyevent)})
        \Bigg)
        \\
        L^{d}[\nparam] &= \Bigg(
            \int (\srdist{\meas{A}}^0 + \srdist{\meas{B}}^0)
            + \int (\crdist{\meas{A}}^0 + \crdist{\meas{B}}^0)
            \\&
            - \sum\limits_{\event \in \meas{A}} \log(1 + {\nparam(\anyevent)})
            - \sum\limits_{\event \in \meas{B}} \log(1 - {\nparam(\anyevent)})
        \Bigg)
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Parameterization by Neural Networks}
    \begin{itemize}
        \item Since $f, g$ should be generic and as expressive as possible, we realize them as NN outputs
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        f(\event) = b_{0} + \sum\limits_{\alpha = 1}^{N} \omega_{\alpha} \sigma(\sum\limits_{j = 1}^{k}\omega^j_{\alpha} \event^j + b_\alpha) \\
        g(\event) = c_{0} + \sum\limits_{\alpha = 1}^{N} \upsilon_{\alpha} \sigma(\sum\limits_{j = 1}^{k}\upsilon^j_{\alpha} \event^j + c_\alpha)
    \end{aligned}
    \end{equation}
    \begin{itemize}
        \item Eta implementation is left for the methodology section, as it's arbitrary and would be furhter manipulated in the future
    \end{itemize}
\end{frame}

\begin{frame}{Wilk's Theorem}
    \begin{theorem}
        \textbf{Wilk theorem}: Given a test statistic $\Lambda$ of a signal hypothesis vs. the null hypothesis, assuming the null hypothesis is true, the negative log-likelihood test statistic converges to a $\chi^2$:
        \begin{equation*}
            -2 \log \Lambda \to \chi^2_r
        \end{equation*}
        with $r$ the number of degrees of freedom of the signal hypothesis minus those of the null one.
    \end{theorem}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item And so, by gathering statistics of $t$, we compare it to a $\chi^2$ distribution and we can estimate how likely it is for the null hypothesis to be true.
        \item When injecting a signal, we'll show the measures of significance:
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        Z_\text{analytic} &= \sqrt{\invllratio} = \sqrt  {
            2 \left( -N_s + \int\limits_{\mathbb{R}^{k}} d\anyevent (N_b(\anyevent) + N_s(\anyevent)) \log \dfrac{N_s(\anyevent) + N_b(\anyevent)}{N_b(\anyevent)} \right)
        }
        \\
        Z_\text{evident} &= \sqrt{\invllratio} = \sqrt{
            2 \left( -N_s + \sum\limits_{\event} d\anyevent (N_b(\event) + N_s(\event)) \log \dfrac{N_s(\event) + N_b(\event)}{N_b(\event)} \right)
        }
        \\
        Z_\text{empirical} &= \Phi^{-1}(1 - \dfrac{\#{t_{bkg} > t_{mean}}}{\#{t_{bkg}}})
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Reference Distribution Choice}
    \begin{itemize}
        \item The natural choice for the reference distribution is the combined sample
        \item We convert integrals over the continuous distribution to normalized sums over the evident sample
    \end{itemize}
    \begin{equation}
        \int h(\anyevent) \refdist^r d\anyevent \to
        \dfrac{1}{N^r}\sum\limits_{\event \in \dataset{N}^r} h(\event)
    \end{equation}
\end{frame}

\begin{frame}{Nuisance Implementation}
    \begin{itemize}
        \item We decided to implement the detector nuisance as binwise constant multiplicative factor for each dimension
    \end{itemize}
    \begin{equation}
        \detector(\anyevent) = \prod\limits_{j=0}^k \detector^j(\anyevent^j)
    \end{equation}
    \begin{equation}
        \detector^j(\anyevent^j) = 
        \begin{cases}
            \detector^{j0}, &x \in \left(-\infty, b_1^j\right) \\
            \detector^{j1}, &x \in \left[b_1^j, b_2^j\right) \\
            ... \\
            \detector^{jn}, &x \in \left[b_{n-1}^j, \infty\right) \\
        \end{cases}
    \end{equation}
    \begin{itemize}
        \item to give
    \end{itemize}
    \begin{equation}
        \nparam(\anyevent) = \prod\limits_{j=0}^{k} \dfrac{1}{2}(\detector^{jh}_\dataset{A} - \detector^{jh}_\dataset{B})
    \end{equation}
\end{frame}

\begin{frame}{t Value Calculation}
    \begin{algorithm}[H]
    \footnotesize
    \caption{Single $t$ value stochastic calculation}
    \label{alg:single_train_algorithm}
    \begin{algorithmic}[1]

    \Statex \textit{Variables in italics are user configurable}
    \Statex % Purely for a clean visual separation

    \Require Configuration of model internal parameters, $\nparam$ usage, dataset source and criteria, number of epochs

    \State Load or generate a sample and randomly divide to $A^{SR}, B^{SR}$ of dimension $\mathit{k}$
    \State Load or generate a sample and randomly divide to $A^{CR}, B^{CR}$ of dimension $\mathit{k}$
    \State $\text{reference}^{SR} \gets A^{SR} + B^{SR}$ and same for $CR$

    \If{\textit{should inject signal}}
        \State Inject the signal to $A^{SR}$
    \EndIf

    \State Numerator model $\gets \text{Model}(f, g, \nparam, \text{loss} = L^{\text{numerator}})$
    \State Denominator model $\gets \text{Model}(f = 0, g = 0, \nparam, \text{loss} = L^{\text{denominator}})$

    \For{\textit{epoch} times}
        \State Train a step for each model
    \EndFor

    \State $t \gets -2 \cdot (\text{numerator model final loss}) + 2 \cdot (\text{denominator model final loss})$

    \State Plot, document and save
    \State \Return $t$

    \end{algorithmic}
    \end{algorithm}
\end{frame}
