\section{Motivation}

\begin{frame}{Classic SM Analysis}
    \begin{itemize}
        \item General steps of a classical search for BSM physics:
        \begin{itemize}
            \item Hypothesize a BSM model
            \item Derive production mechanisms and decay modes
            \item Define event selection
            \item Model background and signal processes
            \item Enhance signal / background ratio by optimizing event selection
            \item Estimate systematic uncertainties
            \item Construct a statistical model
            \item Execute the analysis
        \end{itemize}
        \item The procedure is thorough and labor intensive
        \item Eventually, this examined specific selection(s), out of a huge number of possible ones
    \end{itemize}
\end{frame}

\begin{frame}{DDP}
    \begin{itemize}
        \item We propose an alternative - rapid scanning of a large number of final states
        \item Scan for deviation from what the SM predicts
        \item Direct model builders to explain found signatures
        \item We cope with the look elsewhere effect by correlating signals of similar masses and object combinations
    \end{itemize}
\end{frame}

\section{Overview}

\begin{frame}{LFV - DDP}
    \begin{itemize}
        \item The Standard Model postulates a $SU(3)_C \times SU(2)_L \times U(1)_Y$ local symmetry
        \item In addition to the imposed symmetry group, the SM does not exclude several other gloabl symmetries - those are accidental symmetries
        \item The SM predicts an accidental global symmetry of the lepton flavor universality
        \item There is motivation to look for places in which this symmetry does not hold - driven by observed neutrino oscillations
        \item We'll use this symmetry assumption to compare pairs of equivalent datasets and look for differences without modelling
    \end{itemize}
    \addsizedpic{0.3}{figures/The Standard Model.jpeg}
\end{frame}

\begin{frame}{Our Goal}
    \begin{block}{Goal}
        Given two symmetric datasets, develop a method to estimate whether or not they were created from a same PDF.
        \vspace{0.5cm}

        We'll use it to detect deviations from the SM's predicition of LF universality, with as little as posible assumptions of the backgrounds and singals expected.
        This being said, the method is general and can be used to identify deviations from any symmetry.
    \end{block}
\end{frame}

\begin{frame}{Statistical Fluctuations}

\begin{frame}{Hypothesis Testing}
    \begin{itemize}
        \item Hypothesis formulation, symmterized formalism:
    \end{itemize}
    \begin{block}{null vs signal hypotheses}
        \begin{equation}
        \begin{aligned}
            \nullhypothesis &: \quad
            \begin{cases}
                \pdf_{\dataset{A}}(\eventsymbol)
                &=
                \frac{N_{\dataset{A}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{h(\eventsymbol)} \refdist[\eventsymbol],
                \\
                \pdf_{\dataset{B}}(\eventsymbol)
                &=
                \frac{N_{\dataset{B}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{h(\eventsymbol)+r} \refdist[\eventsymbol],
            \end{cases} \\
            \hypothesis{1} &: \quad
            \begin{cases}
                \pdf_{\dataset{A}}(\eventsymbol)
                &=
                \frac{N_{\dataset{A}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{f(\eventsymbol)} \refdist[\eventsymbol],
                \\
                \pdf_{\dataset{B}}(\eventsymbol)
                &=
                \frac{N_{\dataset{B}}}
                    {\int \refdist[\eventsymbol]\,d\eventsymbol}
                e^{g(\eventsymbol)} \refdist[\eventsymbol].
            \end{cases}
        \end{aligned}
        \end{equation}
    \end{block}
    \begin{itemize}
        \item Test statistic
    \end{itemize}
    \begin{equation}
        t = 2 \log \left(
            \dfrac{
                \max\limits_{f, g} \left(
                    \likelihood(\hypothesis{1}|\dataset{A}, \dataset{B})
                \right)
            }{
                \max\limits_{h, r} \left(
                    \likelihood(\nullhypothesis|\dataset{A}, \dataset{B})
                \right)
            }
        \right)
    \end{equation}
    \begin{itemize}
        \item The maximization is done over each hypothesis space's degrees of freedom by the number of parameters in $f, g$ and $h, r$
    \end{itemize}
\end{frame}

\begin{frame}{t Distribution}
    \begin{itemize}
        \item Wilk's and Wald's theorems tell us that, in the lack of any difference, the distribution of $t$ should be $\chi^2$ with $r$ degrees of freedom, where $r = (\text{\#DoF in signal hypothesis}) - (\text{\#DoF in null hypothesis})$
        \item The farther away the resulting $t$ distribution is from the $\chi^2$, the more likely it is that we have observed a signal
    \end{itemize}

\section{Formulation}

\begin{frame}{Observing Events}
    \begin{itemize}
        \item We consider stochastic physical processes (= collisions)
        \item To probe their behavior, we repeat the same experiment many times and gather statistics of the results
        \item Each collision is an event, $\event$, that is represented as some $k$ physical quantities that we measure - center of mass energy, product $p_t$, etc.
    \end{itemize}
    \begin{equation}
        \event \in \mathbb{R}^k, \quad \event \sim \pdf, \quad \dataset{A} = \{\event\}
    \end{equation}
    \begin{itemize}
        \item A group of observations, $\dataset{A}$, is called a dataset
        \item $\pdf$ is some underlying probability distribution according to which $\event$'s are distributed.
    \end{itemize}
\end{frame}

\begin{frame}{Underlying Probability Distributions}
    \begin{itemize}
        \item \TODO{overlay distributions e,u,refdist}
        \item Looking at pairs of datasets, we express $\pdf$'s relatively to a common reference distribution and a series of shifts:
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        \label{eq:parametrized_distribution}
        \parampdf(\anyevent)^\dataset{A} = e^{\paramf(\anyevent)}(1 + \nparam(\anyevent)) \refdist \\
        \parampdf(\anyevent)^\dataset{B} = e^{\paramg(\anyevent)}(1 - \nparam(\anyevent)) \refdist
    \end{aligned}
    \end{equation}
    \begin{itemize}
        \item the reference distribution, $\refdist$, is composed of the combined datasets
        \item $(1 \pm \nparam(\anyevent))$ represent shifts due to nuisance parameters of the system - in our case, detector effects
        \item $e^{\paramf}, e^{\paramg}$ represent shifts due to potential signal
        \item \TODO{we implement with CR}
    \end{itemize}
\end{frame}

\begin{frame}{Distribution Likelihood}
    \begin{itemize}
        \item this is what we skipped before
        \item Likelihood of a distribution given a dataset $\dataset{A}$ is the probability of observing $\dataset{A}$ had the distribution been true:
    \end{itemize}
    \begin{equation}
        \likelihood(\parampdf | \dataset{A}) = P(\dataset{A} | \parampdf)
    \end{equation}
    \begin{itemize}
        \item Likelihood of a dataset generated from Poisson statistics then becomes
    \end{itemize}
    \begin{equation}
    \label{eq:poisson-likelihood}
        \likelihood(\parampdf | \dataset{A})
        = \dfrac{e^{-N_{\dataset{A}}(\param, \nparam)}}{\abs{\dataset{A}}!}
        \prod\limits_{\event \in \dataset{A}} \parampdf(\event)
        = \dfrac{e^{-\int \parampdf(\anyevent) d\anyevent}}{\abs{\dataset{A}}!}
        \prod\limits_{\event \in \dataset{A}} \parampdf(\event)
    \end{equation}
    \begin{itemize}
        \item which makes the log-likelihood of a distribution like in Eq.~\ref{eq:parametrized_distribution}
    \end{itemize}
    \begin{equation}
    \label{eq:poisson-log-likelihood}
        \llikelihood( \parampdf | \dataset{A}) = -N_{\dataset{A}} + \sum_{\event \in \dataset{A}} \log\big({N}_{\dataset{A}} e^{\paramf(\event)} (1 + \nparam(\event)) \refdist[\event]\big) + C_\dataset{A}
    \end{equation}
\end{frame}

        \item We assume there is no apparent new physics in the CR

\begin{frame}{Hypotheses}
    \begin{itemize}
        \item We postulate two new hypotheses as follows:
    \end{itemize}
    \begin{equation}
    \label{eq:hypothsis-formulation}
    \begin{aligned}
        \nullhypothesis |_{SR} &: \quad
        \begin{cases}
            {\srdist{\dataset{A}}^0} = N_\dataset{A}^{SR} (1 + {\nparam(\anyevent)}) \srrefdist \\
            {\srdist{\dataset{B}}^0} = N_\dataset{B}^{SR} (1 - {\nparam(\anyevent)}) \srrefdist
        \end{cases} \\
        \hypothesis{1} |_{SR} &: \quad
        \begin{cases}
            {\srdist{\dataset{A}}^1} = N_\dataset{A}^{SR} e^{f(\anyevent)} (1 + {\nparam(\anyevent)}) \srrefdist \\
            {\srdist{\dataset{B}}^1} = N_\dataset{B}^{SR} e^{g(\anyevent)} (1 - {\nparam(\anyevent)}) \srrefdist
        \end{cases} \\
        \nullhypothesis |_{CR} = \hypothesis{1} |_{CR} &: \quad
        \begin{cases}
            \crdist{\dataset{A}}^i = N_\dataset{A}^{CR} (1 + {\nparam(\anyevent)}) \crrefdist \\
            \crdist{\dataset{B}}^i = N_\dataset{B}^{CR} (1 - {\nparam(\anyevent)}) \crrefdist
        \end{cases}
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Likelihood and Test Statistics}
    \begin{itemize}
        \item To calculate the log likelihood ratio test statistic
    \end{itemize}
    \begin{equation}
        \Lambda(\param) = \dfrac{
            \likelihood(\hat{\param},\hat{\nparam})
        }{
            \likelihood(\param = 0, \hat{\hat{\nparam}})
        }
    \end{equation}
    \begin{itemize}
        \item We plug the definitions of each hypothesis
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        t &= 2 \log \left(
            \frac{
                \max\limits_{f, g, \nparam} (\likelihood(\hypothesis{1}(\param, \nparam)|\dataset{A}, \dataset{B}))
            }{
                \max\limits_{\nparam} (\likelihood(\nullhypothesis(\nparam)|\dataset{A}, \dataset{B}))
        }\right)
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}{Minimization by Loss Function}
    \begin{itemize}
        \item \TODO{reference the italian paper and why it's good - generic, parameterized}
        \item To estimate $t$ we need to deal with two maximization operation
        \item For that, we define two loss functions
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        L^\text{numerator}[f, g, \nparam] &= -\likelihood(\hypothesis{1}(\paramf, \paramg, \nparam)|\dataset{A}, \dataset{B}) \\
        L^\text{denominator}[\nparam] &= -\likelihood(\nullhypothesis(\nparam)|\dataset{A}, \dataset{B}),
    \end{aligned}
    \end{equation}
    \begin{itemize}
        \item and optimize for each using a simple machine learning model
    \end{itemize}
\end{frame}

\begin{frame}{Parameterization by Neural Networks}
    \begin{itemize}
        \item Since $\paramf, \paramg$ should be as generic and as expressive as possible, we realize them as neural network outputs:
    \end{itemize}
    \addsizedpic{0.5}{
        figures/nn-illustration.png
    }
    \begin{itemize}
        \item This implementation lets us easily control the number of degrees of freedom of our hypothesis space, by controlling the number of neurons in the middle layer
        \item $\nparam$ implementation is left for the methodology section, as it's arbitrary and would be furhter manipulated in the future
    \end{itemize}
\end{frame}

\begin{frame}{Wilk's Theorem}
    \begin{theorem}
        \textbf{Wilk theorem}: Given a test statistic $\Lambda$ of a signal hypothesis vs. the null hypothesis, assuming the null hypothesis is true, the negative log-likelihood test statistic converges to a $\chi^2$:
        \begin{equation*}
            -2 \log \Lambda \to \chi^2_r
        \end{equation*}
        with $r$ the number of degrees of freedom of the signal hypothesis minus those of the null one.
    \end{theorem}
    \begin{itemize}
        \item in calssical analysis f, g has assumptions on the size of a signal
    \end{itemize}
\end{frame}

    \begin{algorithmic}[1]

    \Statex \textit{Variables in italics are user configurable}
    \Statex % Purely for a clean visual separation

    \Require Configuration of model internal parameters, $\nparam$ usage, dataset source and criteria, number of epochs

    \State Load or generate a sample and randomly divide to $A^{SR}, B^{SR}$ of dimension $\mathit{k}$
    \State Load or generate a sample and randomly divide to $A^{CR}, B^{CR}$ of dimension $\mathit{k}$
    \State $\text{reference}^{SR} \gets A^{SR} + B^{SR}$ and same for $CR$

    \If{\textit{should inject signal}}
        \State Inject the signal to $A^{SR}$
    \EndIf

    \State Numerator model $\gets \text{Model}(f, g, \nparam, \text{loss} = L^{\text{numerator}})$
    \State Denominator model $\gets \text{Model}(f = 0, g = 0, \nparam, \text{loss} = L^{\text{denominator}})$

    \For{\textit{epoch} times}
        \State Train a step for each model
    \EndFor

    \State $t \gets -2 \cdot (\text{numerator model final loss}) + 2 \cdot (\text{denominator model final loss})$

    \State Plot, document and save
    \State \Return $t$

    \end{algorithmic}
    \end{algorithm}
\end{frame}

\begin{frame}{Loss Function Development}
    \begin{itemize}
        \item Starting from the required log likelihood ratio expression
    \end{itemize}
        \begin{equation}
        \llikelihood = \dfrac{
            \likelihood(\hat{\param},\hat{\nparam})
        }{
            \likelihood(\param = 0, \hat{\hat{\nparam}})
        }
    \end{equation}
    \begin{itemize}
        \item We plug the signal vs null hypotheses
    \end{itemize}
    \begin{equation}
        \llikelihood = 2 \log \left(
            \frac{
                \max\limits_{f, g, \nparam} (\likelihood(\hypothesis{1}(\param, \nparam)|\dataset{A}, \dataset{B}))
            }{
                \max\limits_{\nparam} (\likelihood(\nullhypothesis(\nparam)|\dataset{A}, \dataset{B}))
        }\right)
    \end{equation}
\end{frame}

\begin{frame}{Loss Function Development}
    \begin{itemize}
        \item Log likelihood can be separated by dataset and region as a sum:
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        \llikelihood(\hypothesis{i} | \dataset{A}, \dataset{B})
        &= \llikelihood(\hypothesis{i}|\dataset{A}^{SR}, \dataset{B}^{SR}) + \llikelihood(\hypothesis{i}|\dataset{A}^{CR}, \dataset{B}^{CR}) \\&
        = \llikelihood(\srdist{\dataset{A}}^i|\dataset{A}^{SR}) + \llikelihood(\srdist{\dataset{B}}^i|\dataset{B}^{SR}) \\&
        + \llikelihood(\crdist{\dataset{A}}^i|\dataset{A}^{CR}) + \llikelihood(\crdist{\dataset{B}}^i|\dataset{B}^{CR})
    \end{aligned}
    \end{equation}
    \begin{itemize}
        \item each of which is governed by poisson statistics as in Eq.~\ref{eq:poisson-log-likelihood} with the corresponding PDF as in Eq.~\ref{eq:hypothsis-formulation}
    \end{itemize}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item This leads us to the full loss expresions:
    \end{itemize}
    \begin{equation}
    \begin{aligned}
        L^{n}[\param, \nparam] &= \Bigg(
            \int (e^{f(\anyevent)}\srdist{\dataset{A}}^0 + e^{g(\anyevent)}\srdist{\dataset{B}}^0)
            + \int (\crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0)
            \\&
            - \sum\limits_{\event \in \dataset{A}^{SR}} f(\anyevent)
            - \sum\limits_{\event \in \dataset{B}^{SR}} g(\anyevent) \\&
            - \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
            - \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
        \Bigg)
        \\
        L^{d}[\nparam] &= \Bigg(
            \int (\srdist{\dataset{A}}^0 + \srdist{\dataset{B}}^0)
            + \int (\crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0)
            \\&
            - \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
            - \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
        \Bigg)
    \end{aligned}
    \end{equation}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item Plugging the discrete datasets in place of the distributions
    \end{itemize}
    \begin{equation}
        \int h(\anyevent) \refdist^r d\anyevent \to
        \dfrac{1}{N^r}\sum\limits_{\event \in \dataset{N}^r} h(\event)
    \end{equation}
    \begin{itemize}
        \item we get
    \end{itemize}
    \begin{equation}
    \label{eq:discrete_loss_functions}
    \begin{aligned}
        L^{n}[\param, \nparam] &= \Bigg(
            \dfrac{1}{N^{SR}} \sum\limits_{\event \in SR} \left(
                {N_\dataset{A}^{SR}} e^{f(\anyevent)} (1 + {\nparam(\anyevent)})
                + {N_\dataset{B}^{SR}} e^{g(\anyevent)} (1 - {\nparam(\anyevent)})
            \right) \\&
            + \dfrac{1}{N^{CR}} \sum\limits_{\event \in CR} \left(
                {N_\dataset{A}^{CR}} (1 + {\nparam(\anyevent)})
                + {N_\dataset{B}^{CR}} (1 - {\nparam(\anyevent)})
            \right)
            \\&
            - \sum\limits_{\event \in \dataset{A}^{SR}} f(\anyevent)
            - \sum\limits_{\event \in \dataset{B}^{SR}} g(\anyevent)
            - \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
            - \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
        \Bigg)
        \\
        L^{d}[\nparam] &= \Bigg(
            \dfrac{1}{N^{SR}} \sum\limits_{\event \in SR} \left(
                {N_\dataset{A}^{SR}} (1 + {\nparam(\anyevent)})
                + {N_\dataset{B}^{SR}} (1 - {\nparam(\anyevent)})
            \right) \\&
            + \dfrac{1}{N^{CR}} \sum\limits_{\event \in CR} \left(
                {N_\dataset{A}^{CR}} (1 + {\nparam(\anyevent)})
                + {N_\dataset{B}^{CR}} (1 - {\nparam(\anyevent)})
            \right)
            \\&
            - \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
            - \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
        \Bigg)
    \end{aligned}
    \end{equation}
\end{frame}
