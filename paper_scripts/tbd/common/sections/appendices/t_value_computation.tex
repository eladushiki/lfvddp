\section{Computing the t value}\label{Appendix_Tvalue}

The equal-sample setting discussed in the main text means equal mean
background event counts at generation, before Poisson fluctuations,
detector effects, and signal injection. The realized observed counts
need not be equal. We therefore retain separate event counts
$N_{\srdataset{A}}$, $N_{\srdataset{B}}$, $N_{\crdataset{A}}$, and
$N_{\crdataset{B}}$ throughout this derivation.

Treating separately the numerator and denominator here, for they will soon require handling on their own

{
\begin{wideequations}
\begin{equation}
\label{eq:overall_test_statistic_numerator}
\begin{aligned}
    \log \nu &= \log \Bigg(
        e^{- \int \srdist{\dataset{A}}^1} \prod\limits_{\event \in \srdataset{A}} \srdistevent{\dataset{A}}^1
        \cdot
        e^{- \int \srdist{\dataset{B}}^1} \prod\limits_{\event \in \srdataset{B}} \srdistevent{\dataset{B}}^1 \\&
        \cdot
        e^{- \int \crdist{\dataset{A}}^0} \prod\limits_{\event \in \crdataset{A}} \crdistevent{\dataset{A}}^0
        \cdot
        e^{- \int \crdist{\dataset{B}}^0} \prod\limits_{\event \in \crdataset{B}} \crdistevent{\dataset{B}}^0
    \Bigg)
    \\
    &= \Bigg(
        - \left(
            \int \srdist{\dataset{A}}^1 + \srdist{\dataset{B}}^1
            + \crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0
        \right)
        \\& + \sum\limits_{\event \in \srdataset{A}} \log({N_\srdataset{A}} (1 + {\nparam(\anyevent)}) (1 + f(\anyevent)) \srrefdist) \\&
        + \sum\limits_{\event \in \srdataset{B}} \log({N_\srdataset{B}}  (1 - {\nparam(\anyevent)}) (1 - f(\anyevent)) \srrefdist)
        \\& + \sum\limits_{\event \in \crdataset{A}} \log({N_\crdataset{A}} (1 + {\nparam(\anyevent)}) \crrefdist) \\&
        + \sum\limits_{\event \in \crdataset{B}} \log({N_\crdataset{B}}  (1 - {\nparam(\anyevent)}) \crrefdist)
    \Bigg)
    \\
    &= \Bigg(
            - \int ((1 + f(\anyevent))\srdist{\dataset{A}}^0 + (1 - f(\anyevent))\srdist{\dataset{B}}^0 )
            - \int (\crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0)
        \\&
        + N_{\srdataset{A}}\log({N_\srdataset{A}})
        + N_{\srdataset{B}}\log({N_\srdataset{B}})
        + N_{\crdataset{A}}\log({N_\crdataset{A}})
        + N_{\crdataset{B}}\log({N_\crdataset{B}})
        \\&
        + \sum\limits_{\event \in \srdataset{A}} [\log(1 + {\nparam(\anyevent)}) + \log(1 + f(\anyevent))]
        \\& + \sum\limits_{\event \in \srdataset{B}} [\log(1 - {\nparam(\anyevent)}) + \log(1 - f(\anyevent))] \\&
        + \sum\limits_{\event \in \crdataset{A}} \log(1 + {\nparam(\anyevent)})
        + \sum\limits_{\event \in \crdataset{B}} \log(1 - {\nparam(\anyevent)})
    \Bigg) + C
\end{aligned}
\end{equation}
\end{wideequations}
}

\noindent Here the parameter-independent reference contribution is
\begin{equation}
    C = \sum_{\event\in\mathrm{SR}}\log \srrefdist[\event]
      + \sum_{\event\in\mathrm{CR}}\log \crrefdist[\event].
\end{equation}
Expanding the product of event intensities gives this sum of logarithms.
The same reference contribution and event-count terms $N_X\log N_X$
appear in both hypotheses and cancel in their log-likelihood difference.
The signal terms do not cancel, since they occur only in the alternative:
\begin{equation*}
    \sum_{\event\in\srdataset{A}}\log(1+f(\event))
    + \sum_{\event\in\srdataset{B}}\log(1-f(\event)).
\end{equation*}
These terms are retained below and in Eq.~\ref{eq:discrete_loss_functions}.

\noindent using the substitutions
\begin{equation*}
\begin{aligned}
\srdist{\dataset{A}}^1 = (1 + f(\anyevent))\srdist{\dataset{A}}^0, \\
\srdist{\dataset{B}}^1 = (1 - f(\anyevent))\srdist{\dataset{B}}^0.
\end{aligned}
\end{equation*}

\begin{wideequations}
\begin{equation}
\label{eq:overall_test_statistic_denominator}
\begin{aligned}
\log \Delta &= \log \Bigg(
    e^{- \int \srdist{\dataset{A}}^0} \prod\limits_{\event \in \srdataset{A}} \srdistevent{\dataset{A}}^0
    \cdot
    e^{- \int \srdist{\dataset{B}}^0} \prod\limits_{\event \in \srdataset{B}} \srdistevent{\dataset{B}}^0 \\&
    \cdot
    e^{- \int \crdist{\dataset{A}}^0} \prod\limits_{\event \in \crdataset{A}} \crdistevent{\dataset{A}}^0
    \cdot
    e^{- \int \crdist{\dataset{B}}^0} \prod\limits_{\event \in \crdataset{B}} \crdistevent{\dataset{B}}^0
\Bigg)
\\&
= \Bigg(
    - \int \left(
        \srdist{\dataset{A}}^0
        + \srdist{\dataset{B}}^0
        + \crdist{\dataset{A}}^0
        + \crdist{\dataset{B}}^0
    \right)
    \\&
    + \sum\limits_{\event \in \srdataset{A}} \log({N_\srdataset{A}} (1 + {\nparam(\anyevent)}) \srrefdist)
    + \sum\limits_{\event \in \srdataset{B}} \log({N_\srdataset{B}} (1 - {\nparam(\anyevent)}) \srrefdist) \\&
    + \sum\limits_{\event \in \crdataset{A}} \log({N_\crdataset{A}} (1 + {\nparam(\anyevent)}) \crrefdist)
    + \sum\limits_{\event \in \crdataset{B}} \log({N_\crdataset{B}} (1 - {\nparam(\anyevent)}) \crrefdist)
\Bigg)
\\&
= \Bigg(
    - \int (\srdist{\dataset{A}}^0 + \srdist{\dataset{B}}^0)
    - \int (\crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0)
    \\&
    + N_{\srdataset{A}}\log({N_\srdataset{A}})
    + N_{\srdataset{B}}\log({N_\srdataset{B}})
    + N_{\crdataset{A}}\log({N_\crdataset{A}})
    + N_{\crdataset{B}}\log({N_\crdataset{B}})
    \\&
    + \sum\limits_{\event \in \srdataset{A}} \log(1 + {\nparam(\anyevent)})
    + \sum\limits_{\event \in \srdataset{B}} \log(1 - {\nparam(\anyevent)}) \\&
    + \sum\limits_{\event \in \crdataset{A}} \log(1 + {\nparam(\anyevent)})
    + \sum\limits_{\event \in \crdataset{B}} \log(1 - {\nparam(\anyevent)})
\Bigg) + C
\end{aligned}
\end{equation}
\end{wideequations}

Plugging back in the expressions cancels some independent terms. From Equations~\ref{eq:high_level_overall_test_statistic}, \ref{eq:overall_test_statistic_numerator} and \ref{eq:overall_test_statistic_denominator}:

\begin{wideequations}
\begin{equation}
\begin{aligned}
\invllratio &= 2 \max\limits_{f, \nparam} \Bigg(
    - \int ((1 + f(\anyevent))\srdist{\dataset{A}}^0 + (1 - f(\anyevent))\srdist{\dataset{B}}^0)
    - \int (\crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0)
    \\&
    + \sum\limits_{\event \in \srdataset{A}} \log (1 + f(\anyevent))
    + \sum\limits_{\event \in \srdataset{B}} \log (1 - f(\anyevent))
    + \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
    + \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
\Bigg)
\\&
- 2 \max\limits_{\nparam} \Bigg(
    - \int (\srdist{\dataset{A}}^0 + \srdist{\dataset{B}}^0)
    - \int (\crdist{\dataset{A}}^0 + \crdist{\dataset{B}}^0)
    \\&
    + \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
    + \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
\Bigg) \\
&= - 2 \min\limits_{f, \nparam} \Bigg(
    \dfrac{1}{N^{SR}} \sum\limits_{\event \in SR} \left(
        N_\srdataset{A} (1 + f(\anyevent)) (1 + {\nparam(\anyevent)})
        + N_\srdataset{B}(1 - f(\anyevent)) (1 - {\nparam(\anyevent)})
    \right)
    \\&
    + \dfrac{1}{N^{CR}} \sum\limits_{\event \in CR} \left(
        N_\crdataset{A} (1 + {\nparam(\anyevent)})
        + N_\crdataset{B} (1 - {\nparam(\anyevent)})
    \right)
    \\&
    - \sum\limits_{\event \in \srdataset{A}} \log (1 + f(\anyevent))
    - \sum\limits_{\event \in \srdataset{B}} \log (1 - f(\anyevent))
    - \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
    - \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
\Bigg)
\\&
+ 2 \min\limits_{\nparam} \Bigg(
    \dfrac{1}{N^{SR}} \sum\limits_{\event \in SR} \left(
        N_\srdataset{A} (1 + {\nparam(\anyevent)})
        + N_\srdataset{B}(1 - {\nparam(\anyevent)})
    \right) \\&
    + \dfrac{1}{N^{CR}} \sum\limits_{\event \in CR} \left(
        N_\crdataset{A} (1 + {\nparam(\anyevent)})
        + N_\crdataset{B}(1 - {\nparam(\anyevent)})
    \right)
    \\&
    - \sum\limits_{\event \in \dataset{A}} \log(1 + {\nparam(\anyevent)})
    - \sum\limits_{\event \in \dataset{B}} \log(1 - {\nparam(\anyevent)})
\Bigg)
\end{aligned}
\end{equation}
\end{wideequations}

\noindent while we substituted the integral with its approximation, the sum over the according domain $\int y(\anyevent) \srrefdist d\anyevent \to \dfrac{1}{N^{SR}} \sum\limits_{\event \in SR} y(\event)$, and similarly for $CR$.
